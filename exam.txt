Write recursive descent parser for grammar:
E -> E+T
E -> T
T -> T*F
T -> F
F -> (E)/id


#include <stdio.h>
#include <ctype.h>

char s[50];
int i = 0;

int E();
int T();
int F();

int E()
{
    if(!T()) return 0;
    while(s[i] == '+')
    {
        i++;
        if(!T()) return 0;
    }
    return 1;
}

int T()
{
    if(!F()) return 0;
    while(s[i] == '*')
    {
        i++;
        if(!F()) return 0;
    }
    return 1;
}

int F()
{
    if(s[i] == '(')
    {
        i++;
        if(E() && s[i] == ')')
        {
            i++;
            return 1;
        }
        return 0;
    }
    if(isalpha(s[i]))
    {
        i++;
        return 1;
    }
    return 0;
}

int main()
{
    printf("Enter expression: ");
    scanf("%49s", s);

    if(E() && s[i] == '\0')
        printf("Valid");
    else
        printf("Invalid");

    return 0;
}

Implement any one storage allocation strategies in stack.

#include <stdio.h>
#define MAX 5

int stack[MAX];
int top = -1;

void push(int value) {
    if (top == MAX - 1) {
        printf("Stack Overflow! Cannot push %d\n", value);
    } else {
        top++;
        stack[top] = value;
        printf("%d pushed into stack\n", value);
    }
}

void pop() {
    if (top == -1) {
        printf("Stack Underflow! Stack is empty\n");
    } else {
        printf("%d popped from stack\n", stack[top]);
        top--;
    }
}

void display() {
    if (top == -1) {
        printf("Stack is empty\n");
    } else {
        printf("Stack elements: ");
        for (int i = top; i >= 0; i--) {
            printf("%d ", stack[i]);
        }
        printf("\n");
    }
}

int main() {
    int choice, value;

    while (1) {
        printf("\n1. Push\n2. Pop\n3. Display\n4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        if (choice == 1) {
            printf("Enter value to push: ");
            scanf("%d", &value);
            push(value);
        } else if (choice == 2) {
            pop();
        } else if (choice == 3) {
            display();
        } else if (choice == 4) {
            printf("Exiting...\n");
            break;
        } else {
            printf("Invalid choice! Try again.\n");
        }
    }

    return 0;
}





. Implementation of Symbol Table and Lexical Analyzer



//Symbol Table

#include <stdio.h>
#include <string.h>

char name[50][20];   // name[i] and type[i] describe symbol i
char type[50][10];
int count = 0;

// returns the index of a symbol, or -1 if not found
int find(char *n) {
    for (int i = 0; i < count; i++)
        if (strcmp(name[i], n) == 0)
            return i;
    return -1;
}

int main() {
    int choice, i;
    char n[20], t[10];

    while (1) {
        printf("\n1.Insert  2.Search  3.Display  4.Exit: ");
        scanf("%d", &choice);

        if (choice == 1) {
            printf("Name and type: ");
            scanf("%19s %9s", n, t);
            if (find(n) != -1)
                printf("Already declared\n");
            else if (count == 50)
                printf("Table full\n");
            else {
                strcpy(name[count], n);
                strcpy(type[count], t);
                count++;
            }
        }
        else if (choice == 2) {
            printf("Name: ");
            scanf("%19s", n);
            i = find(n);
            if (i == -1) printf("Not found\n");
            else         printf("Found: %s %s\n", type[i], name[i]);
        }
        else if (choice == 3) {
            printf("\nIndex  Name  Type\n");
            for (i = 0; i < count; i++)
                printf("%-6d %-5s %s\n", i, name[i], type[i]);
        }
        else
            break;
    }
    return 0;
}







Write a C program to calculate FIRST function for grammar.


#include <stdio.h>
#include <ctype.h>

char lhs[20], rhs[20][10];
int n;

int first(int s, int *set, int *busy)
{
    if(s == '#') return 1;
    if(!isupper(s)) { set[s] = 1; return 0; }
    if(busy[s]) return 0;
    busy[s] = 1;

    int nullable = 0;
    for(int i = 0; i < n; i++)
    {
        if(lhs[i] != s) continue;
        int k = 0;
        while(rhs[i][k] && first(rhs[i][k], set, busy))
            k++;
        if(!rhs[i][k]) nullable = 1;
    }
    busy[s] = 0;
    return nullable;
}

int main()
{
    int i, j, c;
    scanf("%d", &n);

    for(i = 0; i < n; i++)
        scanf(" %c=%9s", &lhs[i], rhs[i]);

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < i && lhs[j] != lhs[i]; j++);
        if(j < i) continue;

        int set[128] = {0}, busy[128] = {0};
        if(first(lhs[i], set, busy)) set['#'] = 1;

        printf("FIRST(%c) = { ", lhs[i]);
        for(c = 0; c < 128; c++)
            if(set[c]) printf("%c ", c);
        printf("}\n");
    }
    return 0;
}







=====C program for Three Address Code.



#include <stdio.h>

char *p;
int t = 1;

void expr(char *res);

void factor(char *res)
{
    if(*p == '(')
    {
        p++;
        expr(res);
        p++;
    }
    else
    {
        res[0] = *p++;
        res[1] = '\0';
    }
}

void term(char *res)
{
    char right[10];
    factor(res);
    while(*p == '*' || *p == '/')
    {
        char op = *p++;
        factor(right);
        printf("t%d = %s %c %s\n", t, res, op, right);
        sprintf(res, "t%d", t++);
    }
}

void expr(char *res)
{
    char right[10];
    term(res);
    while(*p == '+' || *p == '-')
    {
        char op = *p++;
        term(right);
        printf("t%d = %s %c %s\n", t, res, op, right);
        sprintf(res, "t%d", t++);
    }
}

int main()
{
    char infix[50], result[10];

    printf("Enter expression: ");
    scanf("%49s", infix);

    p = infix;
    printf("\nThree Address Code:\n");
    expr(result);

    return 0;
}
